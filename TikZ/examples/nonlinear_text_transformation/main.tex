    do -- See https://tex.stackexchange.com/a/705449
        nodes.rulecodes = {}
        nodes.nuts = node.direct
        nodes.nuts.isglyph = node.direct.is_glyph
        nodes.nuts.effectiveglue = node.direct.effective_glue

        function nodes.nuts.getreplace(n)
            local _, _, h, _, _, t = node.direct.getdisc(n,true)
            return h, t
        end

        require("font-mps")
    end

    -- Global table holding the text transformation functions
    example = {}
    example.transformations = {}

    -- Constants
    local bp_to_sp = tex.sp("1bp")
    local content_box = luatexbase.registernumber("l__example_content_box")
    local expand_scale = 1000000
    local fastcopy = table.fastcopy
    local glyph_id = node.id("glyph")
    local ipairs = ipairs
    local pairs = pairs
    local rule_font = font.define{ name="rule_font" }
    local rule_id = node.id("rule")
    local subdivide_iterations = 2
    local null_flag = -0x40000000


    ------------------------
    --- Helper Functions ---
    ------------------------

    -- Cache the results of a function as a table.
    local function memoized(func)
        return setmetatable({}, { __index = function(cache, key)
            local ret = func(key, cache)
            cache[key] = ret
            return ret
        end })
    end

    -- Replaces a node with another node
    local function replace_node(head, n1, n2)
        local head, current = node.remove(head, n1)
        head, n2 = node.insert_before(head, current, n2)
        return head, n2
    end

    -- Exposes a Lua function as a TeX macro.
    local function register_tex_cmd(name, func, args)
        local scanners = {}
        for _, arg in ipairs(args) do
            scanners[#scanners+1] = token['scan_' .. arg]
        end

        local scanning_func = function()
            local values = {}
            for _, scanner in ipairs(scanners) do
                values[#values+1] = scanner()
            end

            func(table.unpack(values))
        end


        local index = luatexbase.new_luafunction(name)
        lua.get_functions_table()[index] = scanning_func

        token.set_lua(name, index, "global", "protected")
    end

    -- Subdivide segments of a path into smaller segments.
    local function subdivide_segments(inputs)
        local outputs = {}
        local x0, y0 = 0, 0
        for i, input in ipairs(inputs) do
            local command = input[#input]
            if command == "l" then
                -- Convert line to cubic bezier
                local x1, y1 = input[1], input[2]
                local slope_x, slope_y = (x1 - x0) / 3, (y1 - y0) / 3
                outputs[#outputs+1] = {
                    x0 + slope_x, y0 + slope_y,
                    x1 - slope_x, y1 - slope_y,
                    x1, y1,
                    "c"
                }
            elseif command == "c" then
                -- Divide the cubic bezier into two cubic beziers
                local x1, y1 = input[1], input[2]
                local x2, y2 = input[3], input[4]
                local x3, y3 = input[5], input[6]

                x01, y01 = (x0 + x1) / 2, (y0 + y1) / 2
                x12, y12 = (x1 + x2) / 2, (y1 + y2) / 2
                x23, y23 = (x2 + x3) / 2, (y2 + y3) / 2
                x012, y012 = (x01 + x12) / 2, (y01 + y12) / 2
                x123, y123 = (x12 + x23) / 2, (y12 + y23) / 2
                x0123, y0123 = (x012 + x123) / 2, (y012 + y123) / 2

                xa1, ya1 = x01, y01
                xa2, ya2 = x012, y012
                xa3, ya3 = x0123, y0123

                xb1, yb1 = x123, y123
                xb2, yb2 = x23, y23
                xb3, yb3 = x3, y3

                outputs[#outputs+1] = { xa1, ya1, xa2, ya2, xa3, ya3, "c" }
                outputs[#outputs+1] = { xb1, yb1, xb2, yb2, xb3, yb3, "c" }
            elseif command == "m" then
                outputs[#outputs+1] = input
            else
                error("Unknown segment command: " .. tostring(command))
            end

            x0, y0 = input[#input-2], input[#input-1]
        end

        return outputs
    end

    -- Convert rules to fake characters
    local _rule_counter = 0

    local _rule_characters = memoized(function(spec)
        local width, height, depth = spec:match("(%d+),(%d+),(%d+)")
        width, height, depth = tonumber(width), tonumber(height), tonumber(depth)

        _rule_counter = _rule_counter + 1
        return _rule_counter
    end)

    local function rule_character(width, height, depth)
        return rule_font, _rule_characters[("%d,%d,%d"):format(width, height, depth)]
    end

    -- Convert a character to paths
    local char_to_segments = memoized(function(font_id)
        if font_id == rule_font then
            return memoized(function(char_code)
                local spec = table.swapped(_rule_characters)[char_code]
                local width, height, depth = spec:match("(%d+),(%d+),(%d+)")
                width = tonumber(width) / bp_to_sp
                height = tonumber(height) / bp_to_sp
                depth = tonumber(depth) / bp_to_sp

                local segments = {
                    { 0, -depth, "m" },
                    { width, -depth, "l" },
                    { width, height, "l" },
                    { 0, height, "l" },
                    { 0, -depth, "l" }
                }

                -- Rules are longer than character, so we pre-subdivide them
                for _ = 1, subdivide_iterations do
                    segments = subdivide_segments(segments)
                end

                return {
                    width = width,
                    scale = bp_to_sp,
                    segments = segments,
                }
            end)
        end

        local font_characters = fonts.hashes.characters[font_id]
        local shapes = fonts.hashes.shapes[font_id].glyphs
        local tfmdata = font.fonts[font_id]
        local scale = tfmdata.parameters.size / tfmdata.parameters.units

        return memoized(function(char_code)
            local char_index = font_characters[char_code].index
            local shape = shapes[char_index]
            return {
                width = shape.boundingbox[3],
                scale = scale,
                segments = shape.segments,
            }
        end)
    end)

    -- Recursively replace any rules in a node list with fake characters.
    local function replace_rules(head, parent)
        local n = head
        repeat
            if n.id == rule_id then
                if n.width == null_flag then
                    n.width = parent.width
                end
                if n.height == null_flag then
                    n.height = parent.height
                end
                if n.depth == null_flag then
                    n.depth = parent.depth
                end

                local font_id, char_code = rule_character(
                    n.width, n.height, n.depth
                )

                local glyph = node.new(glyph_id)
                glyph.font = font_id
                glyph.char = char_code
                glyph.width = n.width
                glyph.height = n.height
                glyph.depth = n.depth

                head, n = replace_node(head, n, node.hpack(glyph))
            elseif n.list then
                n.list = replace_rules(n.list, n)
            end
            n = n.next
        until not n

        return head
    end


    -------------------------
    --- Drawing Functions ---
    -------------------------

    function example.transformations.identity(x, y)
        return x, y
    end

    -- Replace the internal function used for drawing text outlines
    local transformation_function
    function fonts.metapost.output(_, font_id, char_code, x_offset, y_offset, expand)
        local char = fastcopy(char_to_segments[font_id][char_code])

        -- Subdividing the paths gives better results, inspired by
        -- @mickep's answer at https://tex.stackexchange.com/a/749843
        for _ = 1, subdivide_iterations do
            char.segments = subdivide_segments(char.segments)
        end

        local scale = (char.scale / bp_to_sp)
        local expand_factor = (1 + (expand / expand_scale))
        for _, segment in ipairs(char.segments) do
            for i = 1, (#segment - 1), 2 do
                local x = x_offset + (segment[i] * scale * expand_factor)
                local y = y_offset + (segment[i + 1] * scale)
                x, y = transformation_function(x, y)
                segment[i], segment[i + 1] = x, y
            end
        end

        local mp_code = fonts.metapost.paths(char, 1, 1)
        for i = 1, #mp_code - 1 do
            mp_code[i] = ("nofill(%s)"):format(mp_code[i])
        end
        mp_code[#mp_code] = ("eofill(%s)"):format(mp_code[#mp_code])

        return table.concat(mp_code, "; ") .. "; ", char.width * scale * expand_factor
    end

    -- Draw the text
    register_tex_cmd("__example_transform_text:nnw", function(transform, draw, box)
        box.list = replace_rules(box.list, box)
        tex.box[content_box] = box
        transformation_function = example.transformations[transform]

        local mp_code = fonts.metapost.boxtomp(content_box)
        mp_code = mp_code:gsub("checkbounds%b();", "")

        luamplib.process_mplibcode(([[
            beginfig(0);
                picture p; p := image(%s);
                %s
            endfig;
        ]]):format(mp_code, draw), "")
    end, { "argument", "argument", "list" })
\end{luacode*}

local mm = require "linalg.lua"

-- Store multiple perspective functions
example = { transformations = {} }

for i = 1,100,1 do
  example.transformations["persp"..i] = function(x,y)
    local isp = mm.ISP(x,y)
    local rotated = mm.matrix_multiply(isp, mm.xrotation(-i*2*math.pi/100))
    local sp = mm.stereographic_projection(rotated)
    return sp[1][1], sp[1][2]
  end
end
