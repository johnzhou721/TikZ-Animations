\newif\ifpgf@lowlevelsync
\let\orig@pgflowlevelsynccm\pgflowlevelsynccm
\def\pgflowlevelsynccm{%
  \unless\ifpgf@pt@identity
    \pgf@lowlevelsynctrue
    \pgfgettransform\pgflowlevelsynccmentries
    \begingroup % calculate inverse only once
      \pgftransforminvert\pgfgettransform\pgflowlevelsynccmentriesinverse
      \edef\pgf@marshal{\endgroup\def\noexpand\pgflowlevelsynccmentriesinverse{\pgflowlevelsynccmentriesinverse}}%
    \pgf@marshal
  \fi
  \orig@pgflowlevelsynccm}

\def\pgftransformresetnontranslationslowlevel{%
  \ifpgf@lowlevelsync
    \expandafter\pgftransformresetnontranslationslowlevel@\pgflowlevelsynccmentriesinverse
  \else
    \pgftransformresetnontranslations
  \fi}
\def\pgftransformresetnontranslationslowlevel@#1#2#3#4#5#6{%
  \pgfsettransformentries{#1}{#2}{#3}{#4}{\pgf@pt@x}{\pgf@pt@y}}
\tikzset{low level shift only/.code=\tikz@addtransform{\pgftransformresetnontranslationslowlevel}}
