
\documentclass{beamer}
\beamertemplatenavigationsymbolsempty
\usepackage{
    tikz
    ,tikz-3dplot
}
\pgfmathdeclarefunction{sphereX}{2}{%
    \pgfmathparse{cos(#2)*cos(#1)}%
}
\pgfmathdeclarefunction{sphereY}{2}{%
    \pgfmathparse{cos(#2)*sin(#1)}%
}
\pgfmathdeclarefunction{sphereZ}{2}{%
    \pgfmathparse{sin(#2)}%
}

\pgfmathsetmacro{\polar}{60}
\pgfmathsetmacro{\azimuth}{45}

\pgfmathdeclarefunction{SP}{2}{%
    \pgfmathparse{#1/(1-#2)}%
}

\pgfmathdeclarefunction{loxodromeX}{1}{%
    \pgfmathparse{cos(#1)/sqrt((cos(#1))^2+(sin(#1))^2+(#1/360)^2)}%
}
\pgfmathdeclarefunction{loxodromeY}{1}{%
    \pgfmathparse{sin(#1)/sqrt((cos(#1))^2+(sin(#1))^2+(#1/360)^2)}%
}
\pgfmathdeclarefunction{loxodromeZ}{1}{%
    \pgfmathparse{#1/360/sqrt((cos(#1))^2+(sin(#1))^2+(#1/360)^2)}%
}
\pgfmathsetmacro{\rotation}{0.0}
\begin{document}
    \begin{frame}
    \tdplotsetmaincoords{\polar}{\azimuth}

        \begin{tikzpicture}[tdplot_main_coords]
            \tdplotsetrotatedcoords
                {\rotation}
                {\rotation}
                {\rotation}

            \path[tdplot_screen_coords] 
                (-\textwidth/2,-\textheight/2) rectangle 
                (\textwidth/2,\textheight/2);
            \clip[tdplot_screen_coords] 
                (-\textwidth/2,-\textheight/2) rectangle 
                (\textwidth/2,\textheight/2);

            % lines of latitude on plane
            \foreach \longitude in {0,10,...,350}{
                \tdplottransformrotmain
                    {sphereX(0,\longitude)}
                    {sphereY(0,\longitude)}
                    {sphereZ(0,\longitude)}
                \pgfmathparse{\tdplotresz<0.998}
                \ifnum\pgfmathresult=1
                    \pgfmathsetmacro\lastx{SP(\tdplotresx,\tdplotresz)}
                    \pgfmathsetmacro\lasty{SP(\tdplotresy,\tdplotresz)}
                \fi
                \gdef\opacity{1}
                \foreach \latitude in {0,5,...,355} {
                    \tdplottransformrotmain
                        {sphereX(\latitude,\longitude)}
                        {sphereY(\latitude,\longitude)}
                        {sphereZ(\latitude,\longitude)}
                    \ifdim \tdplotresz pt<0.998pt
                        \pgfmathsetmacro\nextx{SP(\tdplotresx,\tdplotresz)}
                        \pgfmathsetmacro\nexty{SP(\tdplotresy,\tdplotresz)}
                        \draw[opacity=\opacity,ultra thin] 
                            (\lastx,\lasty) -- 
                            (\nextx,\nexty);
                        \gdef\opacity{1}
                        \pgfmathsetmacro\lastx{\nextx}
                        \pgfmathsetmacro\lasty{\nexty}
                        \global\let\lastx\lastx
                        \global\let\lasty\lasty
                    \else
                        \gdef\opacity{0}
                    \fi
                }   
            }

            % lines of longitude on plane
            \foreach \latitude in {0,10,...,350}{
                \tdplottransformrotmain
                    {sphereX(\latitude,0)}
                    {sphereY(\latitude,0)}
                    {sphereZ(\latitude,0)}
                \ifdim \tdplotresz pt<0.998pt
                    \pgfmathsetmacro\lastx{SP(\tdplotresx,\tdplotresz)}
                    \pgfmathsetmacro\lasty{SP(\tdplotresy,\tdplotresz)}
                \fi
                \gdef\opacity{1}
                \foreach \longitude in {0,5,...,355} {
                    \tdplottransformrotmain
                        {sphereX(\latitude,\longitude)}
                        {sphereY(\latitude,\longitude)}
                        {sphereZ(\latitude,\longitude)}
                    \ifdim \tdplotresz pt<0.998pt
                        \pgfmathsetmacro\nextx{SP(\tdplotresx,\tdplotresz)}
                        \pgfmathsetmacro\nexty{SP(\tdplotresy,\tdplotresz)}
                        \draw[opacity=\opacity,ultra thin] (\lastx,\lasty) -- (\nextx,\nexty);
                        \gdef\opacity{1}
                        \pgfmathsetmacro\lastx{\nextx}
                        \pgfmathsetmacro\lasty{\nexty}
                        \global\let\lastx\lastx
                        \global\let\lasty\lasty
                    \else
                        \gdef\opacity{0}
                    \fi
                }
            }

            % projection of loxodrome on plane
            \tdplottransformrotmain
                {loxodromeX(-1800)}
                {loxodromeY(-1800)}
                {loxodromeZ(-1800)}
            \ifdim \tdplotresz pt<0.998pt
                \pgfmathsetmacro\lastx{SP(\tdplotresx,\tdplotresz)}
                \pgfmathsetmacro\lasty{SP(\tdplotresy,\tdplotresz)}
            \fi
            \gdef\opacity{1}
            \foreach \parameter in {-1800,-1790,...,1800}{
                \tdplottransformrotmain
                    {loxodromeX(\parameter)}
                    {loxodromeY(\parameter)}
                    {loxodromeZ(\parameter)}
                \ifdim \tdplotresz pt<0.999pt
                    \pgfmathsetmacro\nextx{SP(\tdplotresx,\tdplotresz)}
                    \pgfmathsetmacro\nexty{SP(\tdplotresy,\tdplotresz)}
                    \draw[opacity=\opacity,very thin,red] 
                        (\lastx,\lasty) -- 
                        (\nextx,\nexty);
                    \gdef\opacity{1}
                    \pgfmathsetmacro\lastx{\nextx}
                    \pgfmathsetmacro\lasty{\nexty}
                    \global\let\lastx\lastx
                    \global\let\lasty\lasty
                \else
                    \gdef\opacity{0}
                \fi
            }

            % This fills the region where we will draw the sphere white.
            \fill[white] (
                {cos(\azimuth)}
                ,{sin(\azimuth)}
            ) arc [
                start angle=\azimuth
                ,end angle={\azimuth-180}
                ,radius=1
            ] -- cycle;
            \fill[
                tdplot_screen_coords
                ,white
            ] (
                {cos(0)}
                ,{sin(0)}
            ) arc [
                start angle=0
                ,end angle=180
                ,radius=1
            ] -- cycle;


            % outline of the sphere
            \draw[tdplot_screen_coords] (
                {cos(0)}
                ,{sin(0)}
            ) arc[
                start angle=0
                ,end angle=180
                ,radius=1
            ];
            \draw arc [
                start angle=\azimuth
                ,end angle={\azimuth-180}
                ,radius=1
            ];



            % longitudinal lines
            \foreach \latitude in {0,10,...,350}{
                \tdplottransformrotmain
                    {sphereX(\latitude,0)}
                    {sphereY(\Vlatitudenn,0)}
                    {sphereZ(\latitude,0)}
                \pgfmathsetmacro\lastx{\tdplotresx}
                \pgfmathsetmacro\lasty{\tdplotresy}
                \pgfmathsetmacro\lastz{\tdplotresz}
                \foreach \longitude in {5,10,...,360}{
                    \tdplottransformrotmain
                        {sphereX(\latitude,\longitude)}
                        {sphereY(\latitude,\longitude)}
                        {sphereZ(\latitude,\longitude)}
                    \pgfmathsetmacro\Vx{\tdplotresx}
                    \pgfmathsetmacro\Vy{\tdplotresy}
                    \pgfmathsetmacro\Vz{\tdplotresz}
                    % Viewing direction components
                    \pgfmathsetmacro\viewdirX{sphereX(-\azimuth,90 - \polar)}
                    \pgfmathsetmacro\viewdirY{sphereY(-\azimuth,90 - \polar)}
                    \pgfmathsetmacro\viewdirZ{sphereZ(-\azimuth,90 - \polar)}

                    % Dot product with the viewer direction
                    \pgfmathsetmacro\currentDotProduct{
                        \lastx*\viewdirX + 
                        \lasty*\viewdirY + 
                        \lastz*\viewdirZ
                    }

                    % Check if the dot product is positive
                    \ifdim\currentDotProduct pt > 0 pt
                    \ifdim\lastz pt>-0.049pt
                    \ifdim\Vz pt>-0.049pt
                        \draw[ultra thin] (\lastx,\lasty,\lastz) -- (\Vx,\Vy,\Vz);
                    \fi\fi\fi
                    \pgfmathsetmacro\lastx{\Vx}
                    \pgfmathsetmacro\lasty{\Vy}
                    \pgfmathsetmacro\lastz{\Vz}
                    \global\let\lastx\lastx
                    \global\let\lasty\lasty
                    \global\let\lastz\lastz
                } % end inner for loop
            } % end outer for loop


            % latitudinal lines
            \foreach \Vnn in {0,10,...,350}{
                \tdplottransformrotmain
                    {sphereX(0,\Vnn)}
                    {sphereY(0,\Vnn)}
                    {sphereZ(0,\Vnn)}
                \pgfmathsetmacro\lastx{\tdplotresx}
                \pgfmathsetmacro\lasty{\tdplotresy}
                \pgfmathsetmacro\lastz{\tdplotresz}
                \foreach \Vn in {5,10,...,360}{
                    \tdplottransformrotmain
                        {sphereX(\Vn,\Vnn)}
                        {sphereY(\Vn,\Vnn)}
                        {sphereZ(\Vn,\Vnn)}
                    \pgfmathsetmacro\Vx{\tdplotresx}
                    \pgfmathsetmacro\Vy{\tdplotresy}
                    \pgfmathsetmacro\Vz{\tdplotresz}
                    % Viewing direction components
                    \pgfmathsetmacro\viewdirX{sphereX(-\azimuth,90 - \polar)}
                    \pgfmathsetmacro\viewdirY{sphereY(-\azimuth,90 - \polar)}
                    \pgfmathsetmacro\viewdirZ{sphereZ(-\azimuth,90 - \polar)}

                    % Dot product with the viewer direction
                    \pgfmathsetmacro\currentDotProduct{
                        \lastx*\viewdirX + 
                        \lasty*\viewdirY + 
                        \lastz*\viewdirZ
                    }

                    % Check if the dot product is positive
                    \ifdim\currentDotProduct pt > -0.3 pt
                    \ifdim\lastz pt>-0.049pt
                    \ifdim\Vz pt>-0.049pt
                        \draw[ultra thin] (\lastx,\lasty,\lastz) -- (\Vx,\Vy,\Vz);
                    \fi\fi\fi
                    \pgfmathsetmacro\lastx{\Vx}
                    \pgfmathsetmacro\lasty{\Vy}
                    \pgfmathsetmacro\lastz{\Vz}
                    \global\let\lastx\lastx
                    \global\let\lasty\lasty
                    \global\let\lastz\lastz
                }
            }



            % loxodrome
            \tdplottransformrotmain
                {loxodromeX(-1800)}
                {loxodromeY(-1800)}
                {loxodromeZ(-1800)}
            \pgfmathsetmacro\lastx{\tdplotresx}
            \pgfmathsetmacro\lasty{\tdplotresy}
            \pgfmathsetmacro\lastz{\tdplotresz}
            \foreach \Vt in {-1800,-1790,...,1800}{
                \tdplottransformrotmain
                    {loxodromeX(\Vt)}
                    {loxodromeY(\Vt)}
                    {loxodromeZ(\Vt)}
                \pgfmathsetmacro\Vx{\tdplotresx}
                \pgfmathsetmacro\Vy{\tdplotresy}
                \pgfmathsetmacro\Vz{\tdplotresz}
                % Viewing direction components
                \pgfmathsetmacro\viewdirX{sphereX(-\azimuth,90 - \polar)}
                \pgfmathsetmacro\viewdirY{sphereY(-\azimuth,90 - \polar)}
                \pgfmathsetmacro\viewdirZ{sphereZ(-\azimuth,90 - \polar)}

                \pgfmathsetmacro\currentDotProduct{
                    \lastx*\viewdirX + 
                    \lasty*\viewdirY + 
                    \lastz*\viewdirZ
                } % Dot product with the viewer direction

                % Check if the dot product is positive
                \ifdim\currentDotProduct pt > -0.3 pt
                \ifdim\lastz pt>-0.049pt
                \ifdim\Vz pt>-0.049pt
                \draw[very thin,red] (\lastx,\lasty,\lastz) -- (\Vx,\Vy,\Vz);
                \fi\fi
                \fi
                \pgfmathsetmacro\lastx{\Vx}
                \pgfmathsetmacro\lasty{\Vy}
                \pgfmathsetmacro\lastz{\Vz}
                \global\let\lastx\lastx
                \global\let\lasty\lasty
                \global\let\lastz\lastz
            }
        \end{tikzpicture}
    \end{frame}
\end{document}
