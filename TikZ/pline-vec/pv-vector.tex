% Draws a dot.
% #1 - options for /tikz/pline-vec/vector/
% #2 - coordinate of the dot 
\newcommand{\pvpoint}[2][]{
    \begingroup 
    \coordinate (@) at (#2);
    \node at (@) [
        circle
        ,minimum size = 2*\pv@vector@pointradius cm
        ,inner sep = 0pt
        ,outer sep = 0pt
        ,fill
        ,#1
    ] {};
    \endgroup
}

% #1 - options for /tikz/pline-vec/vector/
\newcommand{\pvmark}[1][]{
    \begingroup
    \pgfmathsetmacro{\tmp}{\pv@vector@marklength/2}
    \draw
        [
            #1
        ]
        (0,-\tmp pt) -- (0,\tmp pt);
    \endgroup
}

% #1 - options for /tikz/pline-vec/vector/
% #2 - start point
% #3 - displacement from start point
\newcommand{\pvvector}[3][]{
    \begingroup
    \tikzset{#1}
    \coordinate (@start) at (#2);
    \ifnum\pv@vector@points=1
        \pvpoint[#1]{@start}
        \draw
            [
                shorten < = \pv@vector@pointradius cm
                ,{-Stealth[round]}
                ,thick
                ,#1
            ] (@start) -- ++(#3);
    \fi
    \ifnum\pv@vector@points=2
        \pvpoint[#1]{@start}
        \draw
            [
                shorten < = \pv@vector@pointradius cm
                ,shorten > = \pv@vector@pointradius cm
                ,{-Stealth[round]}
                ,thick
                ,#1
            ] (@start) -- ++(#3) coordinate (@end);
        \pvpoint[#1]{@end}
    \fi
    \ifnum\pv@vector@points=3
        \draw
            [
                {-Stealth[round]}
                ,thick
                ,#1
            ] (@start) -- ++(#3);
    \fi
    \ifnum\pv@vector@points=4
        \draw
            [
                shorten > = \pv@vector@pointradius cm
                ,{-Stealth[round]}
                ,thick
                ,#1
            ] (@start) -- ++(#3) coordinate (@end);
        \pvpoint[#1]{@end}
    \fi
    \ifnum\pv@vector@mark=1
        \path
            [
                decoration = {
                    markings
                    ,mark =
                        between positions \pv@vector@markstart
                        and \pv@vector@markend
                        step \pv@vector@markstep
                        with 
                            {
                                \pvmark
                                    [%
                                        #1%
                                        ,-%
                                    ]%
                            }
                }
                ,decorate
            ] (#2) -- ++(#3);
    \fi
    \endgroup
}

% Labels a path:
% #1 - Start Coordinate,
% #2 - End Coordinate,
% #3 - Vector (direction),
% #4 - Relative Position along path,
% #5 - Offset coordinate,
% #6 - The label text.
\newcommand{\pvlabel}[7][]{
    \begingroup
    \path
        [
            decoration = {
                markings
                ,mark =
                    at position #4
                    with 
                        {
                            \node
                                [
                                    #1
                                ]
                                at (#5,#6) {#7};
                        }
            }
            ,decorate
        ] (#2) -- ++(#3);
    \endgroup
}
