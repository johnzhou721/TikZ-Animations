function crossProduct(A, B)
    local cx = A[2] * B[3] - A[3] * B[2]
    local cy = A[3] * B[1] - A[1] * B[3]
    local cz = A[1] * B[2] - A[2] * B[1]
    return {cx, cy, cz}
end
local solenoid = {}
for t = 0, 20*360, 10 do
    local px = 1.5*math.cos(t)
    local py = 1.5*math.sin(t)
    local pz = t/60/40
    local ppx = 1.5*math.cos(t+1)
    local ppy = 1.5*math.sin(t+1)
    local ppz = (t+1)/60/40
    table.insert(solenoid,{px,py,pz,ppx,ppy,ppz})
end
tex.print("\\draw[smooth,samples=300,domain=0:20*360] plot ({1.5*cos(\\x)},{1.5*sin(\\x)},{\\x/60/40});")
for x = -3, 3, 1  do
    --x=0
    tex.print(string.format("\\typeout{X is %f}",x))
    for y = -3, 3, 1  do
        tex.print(string.format("\\typeout{Y is %f}",y))
        for z = -3, 6, 1  do
            local vx = 0
            local vy = 0
            local vz = 0

            for index, value in ipairs(solenoid) do
                local xpx = x-value[1]
                local ypy = y-value[2]
                local zpz = z-value[3]
                local length = math.sqrt((xpx)^2+(ypy)^2+(zpz)^2)

                if length ~= 0 then
                    xpx = xpx/length
                    ypy = ypy/length
                    zpz = zpz/length
                    local pxppx = value[4]-value[1]
                    local pyppy = value[5]-value[2]
                    local pzppz = value[6]-value[3]
                    local CP = crossProduct({pxppx,pyppy,pzppz},{xpx,ypy,zpz})
                    vx = vx + CP[1]/(length)^2
                    vy = vy + CP[2]/(length)^2
                    vz = vz + CP[3]/(length)^2
                end
            end
            vl = math.sqrt(vx^2+vy^2+vz^2)
            if vl ~= 0 then
                tex.print(string.format("\\draw[->,red!%f!blue] (%f,%f,%f) -- ++ (%f,%f,%f);",math.max(0,math.min(vl,100)),x,y,z,0.2*vx/vl,0.2*vy/vl,0.2*vz/vl))
            end
        end
    end
end

\end{luacode}
