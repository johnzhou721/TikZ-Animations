\newcommand{\Point}[1]{%
    % #1  - The coordinate of the point.
    \coordinate (DeliberatelyLongName) at (#1);%
    \node at (DeliberatelyLongName) [%
        circle
        ,minimum size=2*0.062 cm
        ,inner sep=0pt
        ,outer sep=0pt
        ,fill
    ] {};%
}

\newcommand{\Vector}[2]{%
    % #1  - Start coordinate.
    % #2  - Displacement from start coordinate.
    \draw[%
        thick
        ,-{Stealth[round]}
        %,shorten < = 0.062cm
        ,shorten > = 0.062cm
    ] (#1) -- ++(#2);%
}

\newcommand{\Label}[6]{%
    % #1  - Start coordinate.
    % #2  - Displacement from start coordinate.
    % #3  - Position along path.
    % #4  - Relative X-POS.
    % #5  - Relative Y-POS.
    % #6  - The label.
    \path[%
        decoration={%
            markings
            ,mark=at position #3 with {\node at (#4,#5) {#6};},
        },
        postaction={decorate}
    ] (#1) -- ++(#2);%
}

\pgfmathsetmacro{\scale}{0.7}
\tdplotsetmaincoords{60}{130}
\begin{tikzpicture}[tdplot_main_coords,scale=\scale]

    % coordinate axes
    \draw[-Classical TikZ Rightarrow] (0,0) -- (5,0) node[below left]{$x$};
    \draw[-Classical TikZ Rightarrow] (0,0) -- (0,5) node[below right]{$y$};
    \draw[-Classical TikZ Rightarrow] (0,0) -- (0,0,5) node[above right]{$z$};
    
    % normal vector
    \pgfmathsetmacro{\nx}{2}
    \pgfmathsetmacro{\ny}{-1}
    \pgfmathsetmacro{\nz}{5}

    % point in plane
    \pgfmathsetmacro{\px}{2}
    \pgfmathsetmacro{\py}{0}
    \pgfmathsetmacro{\pz}{(-\nx*\px-\ny*\py)/\nz}

    % normalize
    \pgfmathsetmacro{\nl}{sqrt((\nx)^2+(\ny)^2+(\nz)^2)}
    \pgfmathsetmacro{\nx}{\nx/\nl}
    \pgfmathsetmacro{\ny}{\ny/\nl}
    \pgfmathsetmacro{\nz}{\nz/\nl}
    \pgfmathsetmacro{\pl}{sqrt((\px)^2+(\py)^2+(\pz)^2)}
    \pgfmathsetmacro{\px}{\px/\pl}
    \pgfmathsetmacro{\py}{\py/\pl}
    \pgfmathsetmacro{\pz}{\pz/\pl}

    % second point in plane
    \tdplotcrossprod(\nx,\ny,\nz)(\px,\py,\pz)
    
    % We define the plane based off these parameters.
    \coordinate (X) at (\px,\py,\pz);
    \coordinate (Y) at (\tdplotresx,\tdplotresy,\tdplotresz);
    \coordinate (Z) at (\nx,\ny,\nz);
    
    \begin{scope}[x={(X)}, y={(Y)}, z={(Z)}, canvas is xy plane at z=3]
        \fill[opacity=0.3] (-1.75,-2.5) rectangle (2,4);
        \draw (-1.75,-2.5) rectangle (2,4);
        \node[above right] at (-1.75,4) {$\mathcal{P}$};
        \coordinate (A) at (-1,3);
        \coordinate (B) at (0.5,-0.5);
        \coordinate (C) at (0,2.5);
        \pgfmathsetmacro{\recipscale}{1/\scale}
        \begin{scope}[scale=\recipscale]
            \pgflowlevelsynccm
            \Point{-1*\scale,3*\scale}
            \Point{0.5*\scale,-0.5*\scale}
            \Vector{0.5*\scale,-0.5*\scale}{-0.5*\scale,3*\scale}
            \Point{0.5*\scale-0.5*\scale,-0.5*\scale+3*\scale}
        \end{scope}
    \end{scope}

    \Vector{A}{\nx,\ny,\nz}
    \Label{A}{\nx,\ny,\nz}{1}{2ex}{0ex}{$\vec{n}$}
    
    \Vector{B}{\nx,\ny,\nz}
    \Label{B}{\nx,\ny,\nz}{1}{2ex}{0ex}{$\vec{n}$}

    \coordinate (Bn) at ($(B) + (\nx,\ny,\nz)$);
    \draw pic[draw, angle radius=0.2cm]{right angle=C--B--Bn};
    \node[below left] at (B) {$P_0$};
    \node[right] at (C) {$P$};
    \Label{B}{$(C) - (B)$}{0.35}{0ex}{2ex}{$\vec{P_0P}$}

\end{tikzpicture}
