\pgfmathsetmacro{\px}{3*cos(\t)}
\pgfmathsetmacro{\py}{3*sin(\t)}
\pgfmathsetmacro{\pz}{0}
\begin{frame}
\tdplotsetmaincoords{60}{130}
\begin{tikzpicture}[tdplot_main_coords]

    \path[tdplot_screen_coords] (-\textwidth/2,-\textheight/2) rectangle (\textwidth/2,\textheight/2);
    \clip[tdplot_screen_coords] (-\textwidth/2,-\textheight/2) rectangle (\textwidth/2,\textheight/2);

    \begin{scope}[scale=0.75]
    \foreach \mx in {-5,-4,...,5} {
        \foreach \my in {-5,-4,...,5} {
            \foreach \mz in {-5,-4,...,5} {
        
                \pgfmathsetmacro{\pmx}{\mx-\px}
                \pgfmathsetmacro{\pmy}{\my-\py}
                \pgfmathsetmacro{\pmz}{\mz-\pz}
                
                \pgfmathsetmacro{\qmx}{\mx-\qx}
                \pgfmathsetmacro{\qmy}{\my-\qy}
                \pgfmathsetmacro{\qmz}{\mz-\qz}
                
                \pgfmathparse{
                    !(\pmx==0 && \pmy==0 && \pmz==0) && 
                    !(\qmx==0 && \qmy==0 && \qmz==0)
                }
                \ifnum\pgfmathresult=1
                    \pgfmathsetmacro{\pml}{sqrt((\pmx)^2+(\pmy)^2+(\pmz)^2)}
                    \pgfmathsetmacro{\pmx}{\pmx/\pml}
                    \pgfmathsetmacro{\pmy}{\pmy/\pml}
                    \pgfmathsetmacro{\pmz}{\pmz/\pml}
                    \pgfmathsetmacro{\pms}{1/(\pml)^2}
                    \pgfmathsetmacro{\pmx}{\pmx*\pms}
                    \pgfmathsetmacro{\pmy}{\pmy*\pms}
                    \pgfmathsetmacro{\pmz}{\pmz*\pms}
        
                    \pgfmathsetmacro{\qml}{sqrt((\qmx)^2+(\qmy)^2+(\qmz)^2)}
                    \pgfmathsetmacro{\qmx}{\qmx/\qml}
                    \pgfmathsetmacro{\qmy}{\qmy/\qml}
                    \pgfmathsetmacro{\qmz}{\qmz/\qml}
                    \pgfmathsetmacro{\qms}{1/(\qml)^2}
                    \pgfmathsetmacro{\qmx}{\qmx*\qms}
                    \pgfmathsetmacro{\qmy}{\qmy*\qms}
                    \pgfmathsetmacro{\qmz}{\qmz*\qms}
        
                    \pgfmathsetmacro{\fvx}{\pmx-\qmx}
                    \pgfmathsetmacro{\fvy}{\pmy-\qmy}
                    \pgfmathsetmacro{\fvz}{\pmz-\qmz}

                    \pgfmathsetmacro{\fvl}{sqrt((\fvx)^2+(\fvy)^2+(\fvz)^2)}
                    
                    \pgfmathparse{\fvl==0}
                    \ifnum\pgfmathresult=0
                        \pgfmathsetmacro{\fvx}{0.5*\fvx/\fvl}
                        \pgfmathsetmacro{\fvy}{0.5*\fvy/\fvl}
                        \pgfmathsetmacro{\fvz}{0.5*\fvz/\fvl}
                        
                        \draw[->] (\mx,\my,\mz) -- ++(\fvx,\fvy,\fvz);
                    \fi
                \fi
            }
        }
    }

    \foreach \longitude in {0,20,...,160} {
        \foreach \latitude in {0,10,...,350} {
            \draw (
                {\qx+0.3*sphereX(\longitude,\latitude)}
                ,{\qy+0.3*sphereY(\longitude,\latitude)}
                ,{\qz+0.3*sphereZ(\longitude,\latitude)}
            ) -- (
                {\qx+0.3*sphereX(\longitude,\latitude+10)}
                ,{\qy+0.3*sphereY(\longitude,\latitude+10)}
                ,{\qz+0.3*sphereZ(\longitude,\latitude+10)}
            );

            \draw (
                {\px+0.3*sphereX(\longitude,\latitude)}
                ,{\py+0.3*sphereY(\longitude,\latitude)}
                ,{\pz+0.3*sphereZ(\longitude,\latitude)}
            ) -- (
                {\px+0.3*sphereX(\longitude,\latitude+10)}
                ,{\py+0.3*sphereY(\longitude,\latitude+10)}
                ,{\pz+0.3*sphereZ(\longitude,\latitude+10)}
            );
        }
    }

    \foreach \latitude in {0,20,...,160} {
        \foreach \longitude in {0,10,...,350} {
            \draw (
                {\px+0.3*sphereX(\longitude,\latitude)}
                ,{\py+0.3*sphereY(\longitude,\latitude)}
                ,{\pz+0.3*sphereZ(\longitude,\latitude)}
            ) -- (
                {\px+0.3*sphereX(\longitude+10,\latitude)}
                ,{\py+0.3*sphereY(\longitude+10,\latitude)}
                ,{\pz+0.3*sphereZ(\longitude+10,\latitude)}
            );
                
            \draw (
                {\qx+0.3*sphereX(\longitude,\latitude)}
                ,{\qy+0.3*sphereY(\longitude,\latitude)}
                ,{\qz+0.3*sphereZ(\longitude,\latitude)}
            ) -- (
                {\qx+0.3*sphereX(\longitude+10,\latitude)}
                ,{\qy+0.3*sphereY(\longitude+10,\latitude)}
                ,{\qz+0.3*sphereZ(\longitude+10,\latitude)}
            );
        }
    }
    \end{scope}
\end{tikzpicture}
\end{frame}
