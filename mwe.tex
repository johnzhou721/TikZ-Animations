% arara: lualatex
\documentclass[tikz,border = 1cm]{standalone}
\usepackage{tikz-3dplot}
\directlua{require("mwe.lua")}
\begin{document}
\pgfmathsetmacro{\azimuth}{0}
\pgfmathsetmacro{\elevation}{90}
% https://tex.stackexchange.com/a/734566/319072
\pgfmathsetmacro{\CameraX}{sin(\azimuth)*cos(\elevation)}
\pgfmathsetmacro{\CameraY}{-cos(\azimuth)*cos(\elevation)}
\pgfmathsetmacro{\CameraZ}{sin(\elevation)}
\tdplotsetmaincoords{90-\elevation}{\azimuth}
\directlua{observer = {\CameraX,\CameraY,\CameraZ}}
\begin{tikzpicture}[tdplot_main_coords]
    \directlua{
        append_surface(
            0,720,50,
            0,360,9,
            function(u,v) 
                return (2+cosd(1.5*u))*cosd(u)+0.3*sphere_x(u,v) 
            end,
            function(u,v) 
                return (2+cosd(1.5*u))*sind(u)+0.3*sphere_y(u,v)
            end,
            function(u,v) return 2*sind(1.5*u)+0.3*sphere_z(u,v) end
        )
        render_segments()
    }
\end{tikzpicture}
\end{document}
