\tikzset{
    /pline-vec/.is family
    % /pline-vec/ keys
    ,/pline-vec/.cd
    % main orientation key
    ,main orientation/.code n args = {3}{
        \begingroup 
        \pgfmathsetmacro{\xx}{ZYZx(#1,#2,#3,1,0,0)}
        \pgfmathsetmacro{\xy}{ZYZy(#1,#2,#3,1,0,0)}
        \pgfmathsetmacro{\yx}{ZYZx(#1,#2,#3,0,1,0)}
        \pgfmathsetmacro{\yy}{ZYZy(#1,#2,#3,0,1,0)}
        \pgfmathsetmacro{\zx}{ZYZx(#1,#2,#3,0,0,1)}
        \pgfmathsetmacro{\zy}{ZYZy(#1,#2,#3,0,0,1)}
        \edef\temp{
            \noexpand\pgfkeysalso{
                x = {
                    (\xx cm,\xy cm)
                },
                y = {
                    (\yx cm,\yy cm)
                },
                z = {
                    (\zx cm,\zy cm)
                }
            }
        }
        \expandafter
        \endgroup%
        \temp
        % \pgfmathsetmacro
        %     {\observerx}
        %     {pvtransformrotmainx(-#3,-#2,-#1,0,0,1)}
        % \pgfmathsetmacro
        %     {\observery}
        %     {pvtransformrotmainy(-#3,-#2,-#1,0,0,1)}
        % \pgfmathsetmacro
        %     {\observerz}
        %     {pvtransformrotmainz(-#3,-#2,-#1,0,0,1)}
        \pgfmathsetmacro{\pv@internal@a}{(#1)}
        \pgfmathsetmacro{\pv@internal@b}{(#2)}
        \pgfmathsetmacro{\pv@internal@g}{(#3)}
        \directlua{
            observer = pv_ZYZ_rotation_matrix_inverse(
                {
                    \pv@internal@a
                    ,\pv@internal@b
                    ,\pv@internal@g
                }
                ,{0,0,1}
            )
        }
    }
    % boundaries of the coordinate frame
    ,xmin/.estore in = \pv@xmin
    ,xmax/.estore in = \pv@xmax
    ,ymin/.estore in = \pv@ymin
    ,ymax/.estore in = \pv@ymax
    ,zmin/.estore in = \pv@zmin
    ,zmax/.estore in = \pv@zmax
    ,size/.estore in = \pv@size
    % defaults
    ,xmin = -5
    ,xmax = 5
    ,ymin = -5
    ,ymax = 5
    ,zmin = -5
    ,zmax = 5
    ,size = 2
    % /pline-vec/plane/ keys
    ,/pline-vec/plane/.cd
    % normal equation of the plane
    ,na/.estore in = \pv@plane@na
    ,nb/.estore in = \pv@plane@nb
    ,nc/.estore in = \pv@plane@nc
    ,nd/.estore in = \pv@plane@nd
    % defaults
    ,na = 1
    ,nb = 1
    ,nc = 1
    ,nd = 1
    % /pline-vec/line/ keys
    ,/pline-vec/line/.cd
    % point on the line
    ,px/.estore in = \pv@line@px
    ,py/.estore in = \pv@line@py
    ,pz/.estore in = \pv@line@pz
    % a direction vector
    ,dx/.estore in = \pv@line@dx
    ,dy/.estore in = \pv@line@dy
    ,dz/.estore in = \pv@line@dz
    % defaults
    ,px = 1
    ,py = 1
    ,pz = 1
    ,dx = 1
    ,dy = 1
    ,dz = 1
    % /pline-vec/vector/ keys
    ,/pline-vec/vector/.cd
    % radius of points
    ,point radius/.estore in = \pv@vector@pointradius
    % length of mark
    ,mark length/.estore in = \pv@vector@marklength
    % decision to mark or not
    ,mark/.is choice
    ,mark/true/.code = {\edef\pv@vector@mark{1}}
    ,mark/false/.code = {\edef\pv@vector@mark{0}}
    % mark start pos
    ,mark start/.estore in = \pv@vector@markstart
    % mark end pos
    ,mark end/.estore in = \pv@vector@markend
    % mark step length (in cm)
    ,mark step/.estore in = \pv@vector@markstep
    % decision on point placement
    ,points/.is choice
    ,points/behind/.code = {\edef\pv@vector@points{1}}
    ,points/both/.code = {\edef\pv@vector@points{2}}
    ,points/neither/.code = {\edef\pv@vector@points{3}}
    ,points/front/.code = {\edef\pv@vector@points{4}}
    % extra options
    ,options/.style = {#1}
    % defaults
    ,point radius = 0.06
    ,mark length = 4
    ,mark = false
    ,mark start = 0
    ,mark end = 1
    ,mark step = 1
    ,points = behind
    % /pline-vec/z-buffer/ keys
    ,/pline-vec/z-buffer/.cd
    ,u min/.estore in = \pv@zbuffer@umin
    ,u max/.estore in = \pv@zbuffer@umax
    ,u samples/.estore in = \pv@zbuffer@usamples
    ,v min/.estore in = \pv@zbuffer@vmin
    ,v max/.estore in = \pv@zbuffer@vmax
    ,v samples/.estore in = \pv@zbuffer@vsamples
    ,w min/.estore in = \pv@zbuffer@wmin
    ,w max/.estore in = \pv@zbuffer@wmax
    ,w samples/.estore in = \pv@zbuffer@wsamples
    ,u min = 0
    ,u max = 5
    ,u samples = 5
    ,v min = 0
    ,v max = 5
    ,v samples = 5
    ,w min = 0
    ,w max = 5
    ,w samples = 5
}
