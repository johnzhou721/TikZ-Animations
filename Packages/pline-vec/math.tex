% x-component of the cross product of u and v
% #1 - u1
% #2 - u2
% #3 - u3
% #4 - v1
% #5 - v2
% #6 - v3
\pgfmathdeclarefunction{pvcrossproductx}{6}{%
    \pgfmathparse{
        #2 * #6 -
        #3 * #5
    }%
}

% y-component of the cross product of u and v
% #1 - u1
% #2 - u2
% #3 - u3
% #4 - v1
% #5 - v2
% #6 - v3
\pgfmathdeclarefunction{pvcrossproducty}{6}{%
    \pgfmathparse{
        #3 * #4 -
        #1 * #6
    }%
}

% z-component of the cross product of u and v
% #1 - u1
% #2 - u2
% #3 - u3
% #4 - v1
% #5 - v2
% #6 - v3
\pgfmathdeclarefunction{pvcrossproductz}{6}{%
    \pgfmathparse{
        #1 * #5 -
        #2 * #4
    }%
}

% dot product of u and v
% #1 - u1
% #2 - u2
% #3 - u3
% #4 - v1
% #5 - v2
% #6 - v3
\pgfmathdeclarefunction{pvdotproduct}{6}{%
    \pgfmathparse{
        #1 * #4 +
        #2 * #5 +
        #3 * #6
    }%
}

% norm of u
% #1 - u1
% #2 - u2
% #3 - u3
\pgfmathdeclarefunction{pvnorm}{3}{%
    \pgfmathparse{
        sqrt(
            (#1)^2 + 
            (#2)^2 + 
            (#3)^2
        )
    }%
}

% x-component of u after rotatation by ZYZ rotation matrix
% #1 - alpha
% #2 - beta
% #3 - gamma
% #4 - u1
% #5 - u2
% #6 - u3
\pgfmathdeclarefunction{pvtransformrotmainx}{6}{%
    \pgfmathparse{
        (
            cos(#1) * cos(#2) * cos(#3) - 
            sin(#1) * sin(#3)
        ) * #4 +
        (
            -cos(#1) * cos(#2) * sin(#3) - 
            sin(#1) * cos(#3)
        ) * #5 +
        cos(#1) * sin(#2) * #6
    }%
}

% y-component of u after rotatation by ZYZ rotation matrix
% #1 - alpha
% #2 - beta
% #3 - gamma
% #4 - u1
% #5 - u2
% #6 - u3
\pgfmathdeclarefunction{pvtransformrotmainy}{6}{%
    \pgfmathparse{
        (
            sin(#1) * cos(#2) * cos(#3) + 
            cos(#1) * sin(#3)
        ) * #4 +
        (
            -sin(#1) * cos(#2) * sin(#3) + 
            cos(#1) * cos(#3)
        ) * #5 +
        sin(#1) * sin(#2) * #6
    }%
}

% z-component of u after rotatation by ZYZ rotation matrix
% #1 - alpha
% #2 - beta
% #3 - gamma
% #4 - u1
% #5 - u2
% #6 - u3
\pgfmathdeclarefunction{pvtransformrotmainz}{6}{%
    \pgfmathparse{
        -sin(#2) * cos(#3) * #4 +
        sin(#2) * sin(#3) * #5 + 
        cos(#2) * #6
    }%
}

% #1 - longitude
% #2 - latitude
\pgfmathdeclarefunction{pvspherex}{2}{%
  \pgfmathparse{
        cos(#2) *
        cos(#1)
  }%
}

  }%
}

  }%

\newcommand{\pvnormalizevector}[4]{
    \pgfmathsetmacro{\tmpnorm}{norm(#1,#2,#3)}
    \expandafter\pgfmathsetmacro\expandafter{\csname #4x\endcsname}{#1/\tmpnorm}%
    \expandafter\pgfmathsetmacro\expandafter{\csname #4y\endcsname}{#2/\tmpnorm}%
    \expandafter\pgfmathsetmacro\expandafter{\csname #4z\endcsname}{#3/\tmpnorm}%
}
