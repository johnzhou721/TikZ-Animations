\newcommand{\plinepointinplane}[7]{
    \pgfmathsetmacro{\apx}{#1}
    \pgfmathsetmacro{\apy}{#2}
    \pgfmathsetmacro{\apz}{#3}
    \pgfmathsetmacro{\nx}{#4}
    \pgfmathsetmacro{\ny}{#5}
    \pgfmathsetmacro{\nz}{#6}
    \pgfmathsetmacro{\nd}{#7}
    \pgfmathsetmacro{\nl}{sqrt((\nx)^2+(\ny)^2+(\nz)^2)}
    \pgfmathsetmacro{\zpx}{\nx/\nl}
    \pgfmathsetmacro{\zpy}{\ny/\nl}
    \pgfmathsetmacro{\zpz}{\nz/\nl}
    \pgfmathsetmacro{\ipx}{0}
    \pgfmathsetmacro{\ipy}{0}
    \pgfmathsetmacro{\ipz}{0}
    \pgfmathsetmacro{\ipl}{(\nx*\ipx + \ny*\ipy + \nz*\ipz - \nd)/(\nl)^2}
    \pgfmathsetmacro{\ipx}{\ipx-\ipl*\nx}
    \pgfmathsetmacro{\ipy}{\ipy-\ipl*\ny}
    \pgfmathsetmacro{\ipz}{\ipz-\ipl*\nz}
    \pgfmathsetmacro{\ppx}{\nx+1}
    \pgfmathsetmacro{\ppy}{\ny}
    \pgfmathsetmacro{\ppz}{\nz}
    \tdplotcrossprod(\ppx,\ppy,\ppz)(\nx,\ny,\nz)
    \pgfmathsetmacro{\ppx}{\tdplotresx}
    \pgfmathsetmacro{\ppy}{\tdplotresy}
    \pgfmathsetmacro{\ppz}{\tdplotresz}
    \pgfmathsetmacro{\xpx}{\ppx-\ipx}
    \pgfmathsetmacro{\xpy}{\ppy-\ipy}
    \pgfmathsetmacro{\xpz}{\ppz-\ipz}
    \pgfmathsetmacro{\xpl}{sqrt((\xpx)^2 + (\xpy)^2 + (\xpz)^2)}
    \pgfmathsetmacro{\xpx}{\xpx/\xpl}
    \pgfmathsetmacro{\xpy}{\xpy/\xpl}
    \pgfmathsetmacro{\xpz}{\xpz/\xpl}
    \tdplotcrossprod(\nx,\ny,\nz)(\xpx,\xpy,\xpz)
    \pgfmathsetmacro{\ypx}{\tdplotresx}
    \pgfmathsetmacro{\ypy}{\tdplotresy}
    \pgfmathsetmacro{\ypz}{\tdplotresz}
    \pgfmathsetmacro{\ypl}{sqrt((\ypx)^2 + (\ypy)^2 + (\ypz)^2)}
    \pgfmathsetmacro{\ypx}{\ypx/\ypl}
    \pgfmathsetmacro{\ypy}{\ypy/\ypl}
    \pgfmathsetmacro{\ypz}{\ypz/\ypl}
    \pgfmathsetmacro{\vpx}{\apx-\ipx}
    \pgfmathsetmacro{\vpy}{\apy-\ipy}
    \pgfmathsetmacro{\vpz}{\apz-\ipz}
    % output below
    \pgfmathsetmacro{\bpx}{-(\vpx*\xpx + \vpy*\xpy + \vpz*\xpz)}
    \pgfmathsetmacro{\bpy}{-(\vpx*\ypx + \vpy*\ypy + \vpz*\ypz)}
}
