\directlua{require("pv-plane.lua")}

\newcommand{\pvgetplane}[2][]{
    \begingroup
    \tikzset{/pline-vec/plane/.cd,#1}
    \edef\temp{
    \noexpand%
    \directlua{
        pv_get_plane(
            \pv@size
            ,{\pv@plane@na,\pv@plane@nb,\pv@plane@nc}
            ,\pv@plane@nd
            ,\pv@xmin,\pv@xmax
            ,\pv@ymin,\pv@ymax
            ,\pv@ymin,\pv@ymax
            ,"#2"
        )
    }}
    \expandafter
    \endgroup%
    \temp
}

\newcommand{\pvgetrectangularprism}[1][]{%
    \tikzset{/pline-vec/plane/.cd,#1}%

    \global\def\backfaces{}%
    \global\def\frontfaces{}%

    \setPoint{\pv@xmin}{\pv@ymin}{\pv@zmin}{o}%
    \setPoint{\pv@xmax}{\pv@ymin}{\pv@zmin}{x}%
    \setPoint{\pv@xmin}{\pv@ymax}{\pv@zmin}{y}%
    \setPoint{\pv@xmax}{\pv@ymax}{\pv@zmin}{xy}%
    \setPoint{\pv@xmin}{\pv@ymin}{\pv@zmax}{oz}%
    \setPoint{\pv@xmax}{\pv@ymin}{\pv@zmax}{xz}%
    \setPoint{\pv@xmin}{\pv@ymax}{\pv@zmax}{yz}%
    \setPoint{\pv@xmax}{\pv@ymax}{\pv@zmax}{xyz}%

    % Define helper macro using \def and doubled ## for parameters
    \def\appendface##1##2##3{%
        \pgfmathparse{##2*(##1) >= 0}%
        \ifnum\pgfmathresult=1
            \edef\backfaces{\unexpanded\expandafter{\frontfaces}##3}%
        \else
            \edef\frontfaces{\unexpanded\expandafter{\backfaces}##3}%
        \fi
    }%

    % Each face (note that coordinates like \ox, \oy, \oz must be defined by \setPoint)
    \appendface{\observerz}{-1}{%
        \noexpand\draw (\ox,\oy,\oz) -- (\xx,\xy,\xz) -- (\xyx,\xyy,\xyz) -- (\yx,\yy,\yz) -- cycle;
    }%

    \appendface{\observerz}{1}{%
        \noexpand\draw (\ozx,\ozy,\ozz) -- (\xzx,\xzy,\xzz) -- (\xyzx,\xyzy,\xyzz) -- (\yzx,\yzy,\yzz) -- cycle;
    }%

    \appendface{\observerx}{-1}{%
        \noexpand\draw (\ox,\oy,\oz) -- (\yx,\yy,\yz) -- (\yzx,\yzy,\yzz) -- (\ozx,\ozy,\ozz) -- cycle;
    }%

    \appendface{\observerx}{1}{%
        \noexpand\draw (\xx,\xy,\xz) -- (\xyx,\xyy,\xyz) -- (\xyzx,\xyzy,\xyzz) -- (\xzx,\xzy,\xzz) -- cycle;
    }%

    \appendface{\observery}{-1}{%
        \noexpand\draw (\ox,\oy,\oz) -- (\xx,\xy,\xz) -- (\xzx,\xzy,\xzz) -- (\ozx,\ozy,\ozz) -- cycle;
    }%

    \appendface{\observery}{1}{%
        \noexpand\draw (\yx,\yy,\yz) -- (\xyx,\xyy,\xyz) -- (\xyzx,\xyzy,\xyzz) -- (\yzx,\yzy,\yzz) -- cycle;
    }%
}
