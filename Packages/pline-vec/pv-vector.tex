% Draws a dot.
% #1 - options for /tikz/pline-vec/vector/
% #2 - coordinate of the dot 
\newcommand{\pvdot}[2][]{
    \begingroup
    \pgfkeys{
        /pline-vec/vector/.cd
        ,#1
    }
    \coordinate (tmp) at (#2);
    \node 
        at (tmp) 
        [
            circle
            ,minimum size = 2 * \pv@vector@pointradius cm
            ,inner sep = 0pt
            ,outer sep = 0pt
            ,fill
            ,/pline-vec/vector/.cd
            ,options
        ] {};
    \endgroup
}

% #1 - options for /tikz/pline-vec/vector/
\newcommand{\pvmark}[1][]{
    \begingroup
    \tikzset{
        /pline-vec/vector/.cd
        ,#1
    }
    \pgfmathsetmacro{\tmp}{\pv@vector@marklength/2}
    \draw
        [
            /pline-vec/vector/.cd
            ,options
        ]
        (0,-\tmp pt) -- (0,\tmp pt);
    \endgroup
}

% #1 - options for /tikz/pline-vec/vector/
% #2 - start point
% #3 - displacement from start point
\newcommand{\pvvector}[3][]{
    \begingroup
    \tikzset{
        /pline-vec/vector/.cd
        ,#1
    }
    \coordinate (@start) at (#2);
    \ifnum\pv@vector@points=1
        \pvdot{@start}
        \draw
            [
                shorten < = \pv@vector@pointradius cm
                ,/pline-vec/vector/.cd
                ,options
            ] (@start) -- ++(#3);
    \fi
    \ifnum\pv@vector@points=2
        \pvdot{@start}
        \draw
            [
                shorten < = \pv@vector@pointradius cm
                ,shorten > = \pv@vector@pointradius cm
                ,/pline-vec/vector/.cd
                ,options
            ] (@start) -- ++(#3) coordinate (@end);
        \pvdot{@end}
    \fi
    \ifnum\pv@vector@points=3
        \draw
            [
                /pline-vec/vector/.cd
                ,options
            ] (@start) -- ++(#3);
    \fi
    \ifnum\pv@vector@points=4
        \draw
            [
                shorten > = \pv@vector@pointradius cm
                ,/pline-vec/vector/.cd
                ,options
            ] (@start) -- ++(#3) coordinate (@end);
        \pvdot{@end}
    \fi
    \ifnum\pv@vector@mark=1
        \path
            [
                decoration = {
                    markings
                    ,mark =
                        between positions \pv@vector@markstart
                        and \pv@vector@markend
                        step = \pv@vector@markstep
                        with 
                            {
                                \pvmark
                                    [
                                        /pline-vec/vector/.cd
                                        ,options
                                    ]
                            }
                }
                ,decorate
            ] (#2) -- ++(#3);
    \fi
    \endgroup
}

% Labels a path:
% #1 - Start Coordinate,
% #2 - End Coordinate,
% #3 - Vector (direction),
% #4 - Relative Position along path,
% #5 - Offset coordinate,
% #6 - The label text.
\newcommand{\pvlabel}[7][]{
    \begingroup
    \tikzset{
        /pline-vec/vector/.cd
        ,#1
    }
    \path
        [
            decoration = {
                markings
                ,mark =
                    at position #4
                    with 
                        {
                            \node
                                [
                                    /pline-vec/vector/.cd
                                    ,options
                                ]
                                at (#5,#) {#7};
                        }
            }
            ,decorate
        ] (#2) -- ++(#3);
    \endgroup
  \begingroup%
    \pgfqkeys{/pvVector}{#1}%
    \path[
      decoration={
        markings,
        mark=at position #4 with {\node[/pvVector/options aux] at (#5,#6) {#7};},
      },
      postaction={decorate}
    ]
    (#2) -- ++(#3);
  \endgroup
}
