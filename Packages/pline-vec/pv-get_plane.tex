\newcommand{\plinepointinplane}[4]{
    \begingroup
    \pgfqkeys{/pline}{#1}
    \setPoint{#2}{#3}{#4}{ap}
    \pgfmathsetmacro{\nd}{\pline@nod}
    \normalizeVector{\pline@noa}{\pline@nob}{\pline@noc}{zp}
    \setPoint{0}{0}{0}{ip}
    \pgfmathsetmacro{\ipl}{(\nx*\ipx + \ny*\ipy + \nz*\ipz - \nd)/(\nl)^2}
    \setPoint{\ipx-\ipl*\nx}{\ipy-\ipl*\ny}{\ipz-\ipl*\nz}{ip}
    \orthogonalVector{\ipx}{\ipy}{\ipz}{pp}
    \crossProduct{\ppx}{\ppy}{\ppz}{\nx}{\ny}{\nz}{pp}
    \normalizeVector{\ppx-\ipx}{\ppy-\ipy}{\ppz-\ipz}{xp}
    \crossProduct{\nx}{\ny}{\nz}{\xpx}{\xpy}{\xpz}{yp}
    \normalizeVector{\ypx}{\ypy}{\ypz}{yp}
    \setPoint{\apx-\ipx}{\apy-\ipy}{\apz-\ipz}{vp}
    % output below
    \pgfmathsetmacro{\bpx}{-(\vpx*\xpx + \vpy*\xpy + \vpz*\xpz)}
    \pgfmathsetmacro{\bpy}{-(\vpx*\ypx + \vpy*\ypy + \vpz*\ypz)}
    \aftergroup\let\aftergroup\plinebpx\aftergroup\bpx
    \aftergroup\let\aftergroup\plinebpy\aftergroup\bpy
    \endgroup
}


% Takes two (scaled) points in a SCALED coordinate system (x, y and z)
% and outputs the 
\newcommand{\connectPoints}[7]{
    \begingroup
    % set points one and two
    \setPoint{#1}{#2}{#2}{po}
    \setPoint{#4}{#5}{#6}{pt}
    \setPoint{\ptx-\pox}{\pty-\poy}{\ptz-\poz}{d}
    % length between points 1 and 2
    \pgfmathsetmacro{\length}{norm(\dx,\dy,\dz)}
    \normalizeVector{\dx}{\dy}{\dz}{x}
    \setPoint{\plinescalex*\pline@nx}{\plinescaley*\pline@ny}{\plinescalez*\pline@nz}{z}
    \crossProduct{\dx}{\dy}{\dz}{\zx}{\zy}{\zz}{y}
    \begin{scope}[
        x = {(\xx,\xy,\xz)}
        ,y = {(\yx,\yy,\yz)}
        ,z = {(\zx,\zy,\zz)}
        ,shift = {(\pox,\poy,\poz)}
        ,canvas is xy plane at z = 0
    ]
        \pgflowlevelsynccm
        \pvVector[#7]{0,0}{\length,0} 
    \end{scope}
    \endgroup
}

\newcommand{\rotatedDot}[4]{
    \begingroup
    \setPoints
        {\plinescalex*#1}
        {\plinescaley*#2}
        {\plinescalez*#2}
        {po}
    \setPoints
        {\plinescalex*\plinenx}
        {\plinescaley*\plineny}
        {\plinescalez*\plinenz}
        {n}
    \normalizeVector{\nx}{\ny}{\nz}{n}
    \orthogonalVector{\nx}{\ny}{\nz}{o}
    \crossProduct{\nx}{\ny}{\nz}{\ox}{\oy}{\oz}{o}
    \normalizeVector{\ox}{\oy}{\oz}{o}
    \crossProduct{\nx}{\ny}{\nz}{\ox}{\oy}{\oz}{t}
    \normalizeVector{\tx}{\ty}{\tz}{t}
    \tikzset{
        plane/.style = {
            x = {(\ox,\oy,\oz)}
            ,y = {(\tx,\ty,\tz)}
            ,z = {(\nx,\ny,\nz)}
            ,shift = {(\pox,\poy,\poz)}
            ,canvas is xy plane at z = 0
        }
    }
    \begin{scope}[plane]
        \pgflowlevelsynccm
        \pvDot[#4]{0,0}; 
    \end{scope}
    \endgroup
}
