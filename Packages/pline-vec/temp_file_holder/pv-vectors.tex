% Define the main keys for pvVector.
\pgfqkeys{/pvVector}{
    .is family,
    color/.store in = \pvVector@color,
    color=black,
    mark/.is choice,
    mark/false/.code={\def\pvVector@mark{false}},
    mark/true/.code={\def\pvVector@mark{true}},
    mark start/.store in = \pvVector@markStart,
    mark stop/.store in = \pvVector@markStop,
    mark step/.store in = \pvVector@markStep,
    mark start=0,
    mark stop=1,
    mark step=1cm,
    mark=false,
    dots/.is choice,
    dots/neither/.code={\def\pvVector@dots{neither}},
    dots/behind/.code={\def\pvVector@dots{behind}},
    dots/front/.code={\def\pvVector@dots{front}},
    dots/both/.code={\def\pvVector@dots{both}},
    dots=behind,
    % Auxiliary options style for vector (and its children)
    options aux/.style = {},
    options/.code = {\pgfqkeys{/pvVector}{options aux/.style = {#1},}},
}

% Define a subfamily for pvDot as a child of pvVector.
\pgfqkeys{/pvVector/dot}{
    .is family,
    % Auxiliary options style for dot.
    options aux/.style = {},
    options/.code = {\pgfqkeys{/pvVector/dot}{options aux/.style = {#1},}},
}

% A general helper conditional.
\newcommand\pvIf[2]{%
    \begingroup
      \def\tmp{#2}%
      \expandafter
    \endgroup
    \ifx#1\tmp
      \expandafter\@firstoftwo
    \else
      \expandafter\@secondoftwo
    \fi
}

% Size factor for the dot
\pgfmathsetmacro\PtR{0.06}

% Redefine \pvDot to use the /pvVector/dot subfamily and inherit the color.
\newcommand{\pvDot}[2][]{%
  \begingroup%
    \pgfqkeys{/pvVector/dot}{#1}%
    \coordinate (tmp) at (#2);
    \node at (tmp) [
      circle,
      minimum size=2*\PtR cm,
      inner sep=0pt,
      outer sep=0pt,
      fill=\pvVector@color, % Inherit parent's color
      /pvVector/dot/options aux
    ] {};
  \endgroup
}

% Define \pvMark for drawing marks on vectors.
\newcommand{\pvMark}[1][]{%
  \begingroup%
    \pgfqkeys{/pvVector}{#1}%
    \draw[thick,\pvVector@color] (0,-2pt) -- (0,2pt);
  \endgroup
}

% Define the vector command; now calls \pvDot without an explicit fill.
\newcommand{\pvVector}[3][]{%
  \begingroup
    \pgfqkeys{/pvVector}{#1}%
    \coordinate (pvVectorStart) at (#2);
    \coordinate (pvVectorEnd) at (#3);
    \pvIf\pvVector@dots{neither}{%
      \draw[thick, -{Stealth[round]},\pvVector@color,/pvVector/options aux] 
        (pvVectorStart) -- ++(pvVectorEnd);
    }{}%
    \pvIf\pvVector@dots{behind}{%
      \pvDot{pvVectorStart}%
      \draw[\pvVector@color,thick, -{Stealth[round]}, shorten <=\PtR cm,/pvVector/options aux] 
        (pvVectorStart) -- ++(pvVectorEnd);
    }{}%
    \pvIf\pvVector@dots{front}{%
      \draw[thick, -{Stealth[round]}, shorten >=\PtR cm,\pvVector@color,/pvVector/options aux] 
        (pvVectorStart) -- ++(pvVectorEnd) coordinate(DeliberatelyLongName);
      \pvDot{DeliberatelyLongName}%
    }{}%
    \pvIf\pvVector@dots{both}{%
      \pvDot{pvVectorStart}%
      \draw[thick, -{Stealth[round]}, shorten <=\PtR cm, shorten >=\PtR cm,\pvVector@color,/pvVector/options aux] 
        (pvVectorStart) -- ++(pvVectorEnd) coordinate(DeliberatelyLongName);
      \pvDot{DeliberatelyLongName}%
    }{}%
    \pvIf\pvVector@mark{true}{%
      \path[
        decoration={markings,
          mark=between positions \pvVector@markStart and \pvVector@markStop step \pvVector@markStep with {\pvMark},
        },
        postaction={decorate}
      ]
      (#2) -- ++(#3);
    }{}%
  \endgroup
}

% Define a helper command to label a path (unchanged).
\newcommand{\LabelPath}[7][]{%
  \begingroup%
    \pgfqkeys{/pvVector}{#1}%
    % Labels a path:
    % #1 - Start Coordinate,
    % #2 - End Coordinate,
    % #3 - Vector (direction),
    % #4 - Relative Position along path,
    % #5 - Offset coordinate,
    % #6 - The label text.
    \path[
      decoration={
        markings,
        mark=at position #4 with {\node[/pvVector/options aux] at (#5,#6) {#7};},
      },
      postaction={decorate}
    ]
    (#2) -- ++(#3);
  \endgroup
}
