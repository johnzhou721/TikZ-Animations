 % These functions define the cross product, which accepts 6 numbers 
 % and outputs 3.
\pgfmathdeclarefunction{crossX}{6}{%
    \pgfmathparse{#2*#6-#3*#5}%
}
\pgfmathdeclarefunction{crossY}{6}{%
    \pgfmathparse{#3*#4-#1*#6}%
}
\pgfmathdeclarefunction{crossZ}{6}{%
    \pgfmathparse{#1*#5-#2*#4}%
}
% Defines the dot product
\pgfmathdeclarefunction{dotproduct}{6}{%
    \pgfmathparse{#1*#4+#2*#5+#3*#6}%
}
% defines the norm formula in R3
\pgfmathdeclarefunction{norm}{3}{%
    \pgfmathparse{sqrt((#1)^2+(#2)^2+(#3)^2)}%
}
% These functions define the ZYZ Euler Angle rotation matrix, which 
% inputs 6 numbers and outputs 3. This particular matrix is 
% complementary to tikz-3dplot.
% https://tex.stackexchange.com/a/736767/319072
\pgfmathdeclarefunction{transformrotmainX}{6}{%
    % #1  - Z rotation
    % #2  - Y rotation
    % #3  - Z rotation
    % #4  - x value
    % #5  - y value
    % #6  - z value
    \pgfmathparse{
        (
            (
                sin(#1) * cos(#2) * cos(#3) + 
                cos(#1) * sin(#3)
            ) * #4 +
            (
                -sin(#1) * cos(#2) * sin(#3) + 
                cos(#1) * cos(#3)
            ) * #5 +
            sin(#1) * sin(#2) * #6
        ) * -1
    }%
}
\pgfmathdeclarefunction{transformrotmainY}{6}{%
    \pgfmathparse{
        (
            cos(#1) * cos(#2) * cos(#3) - 
            sin(#1) * sin(#3)
        ) * #4 +
        (
            -cos(#1) * cos(#2) * sin(#3) - 
            sin(#1) * cos(#3)
        ) * #5 +
        cos(#1) * sin(#2) * #6
    }%
}
\pgfmathdeclarefunction{transformrotmainZ}{6}{%
    \pgfmathparse{
        -sin(#2) * cos(#3) * #4 +
        sin(#2) * sin(#3) * #5 + 
        cos(#2) * #6
    }%
}
% These functions define the parametric equation for a sphere.
\pgfmathdeclarefunction{sphereX}{2}{%
  \pgfmathparse{cos(#2)*cos(#1)}%
}
\pgfmathdeclarefunction{sphereY}{2}{%
  \pgfmathparse{cos(#2)*sin(#1)}%
}
\pgfmathdeclarefunction{sphereZ}{2}{%
  \pgfmathparse{sin(#2)}%
}
% Normalize a 3D vector
\newcommand{\normalizeVector}[4]{% #1,#2,#3 -> output prefix #4
  \pgfmathsetmacro{\tmpnorm}{norm(#1,#2,#3)}%
  \expandafter\pgfmathsetmacro\expandafter{\csname #4x\endcsname}{#1/\tmpnorm}%
  \expandafter\pgfmathsetmacro\expandafter{\csname #4y\endcsname}{#2/\tmpnorm}%
  \expandafter\pgfmathsetmacro\expandafter{\csname #4z\endcsname}{#3/\tmpnorm}%
}

% Compute cross product of two vectors
\newcommand{\crossProduct}[7]{% u=(#1,#2,#3), v=(#4,#5,#6), out prefix = #7
  \expandafter\pgfmathsetmacro\expandafter{\csname #7x\endcsname}{crossX(#1,#2,#3,#4,#5,#6)}%
  \expandafter\pgfmathsetmacro\expandafter{\csname #7y\endcsname}{crossY(#1,#2,#3,#4,#5,#6)}%
  \expandafter\pgfmathsetmacro\expandafter{\csname #7z\endcsname}{crossZ(#1,#2,#3,#4,#5,#6)}%
}

% Set a 3D point
\newcommand{\setPoint}[4]{% (x, y, z) -> prefix
  \expandafter\pgfmathsetmacro\expandafter{\csname #4x\endcsname}{#1}%
  \expandafter\pgfmathsetmacro\expandafter{\csname #4y\endcsname}{#2}%
  \expandafter\pgfmathsetmacro\expandafter{\csname #4z\endcsname}{#3}%
}

\newcommand{\orthogonalVector}[4]{% #1,#2,#3 -> output prefix #4
    \pgfmathparse{#1!=0 && #2==0 && #2==0}
    \ifnum\pgfmathresult=1
        \setPoint{
            crossX(#1,#2,#3,0,1,0)
        }{
            crossY(#1,#2,#3,0,1,0)
        }{
            crossZ(#1,#2,#3,0,1,0)
        }{#4}
    \else
        \setPoint{
            crossX(#1,#2,#3,1,0,0)
        }{
            crossY(#1,#2,#3,1,0,0)
        }{
            crossZ(#1,#2,#3,1,0,0)
        }{#4}
    \fi
}
