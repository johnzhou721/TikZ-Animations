\pgfqkeys{/mainCoordinateFrame}{
    .is family
    ,alpha/.store in = \MCF@a
    ,beta/.store in = \MCF@b
    ,gamma/.store in = \MCF@g
    ,alpha = 0
    ,beta = 0
    ,gamma = 0
}
\newcommand{\setMainOrientation}[1][]{
  % https://tex.stackexchange.com/q/742150/319072
  \pgfqkeys{/mainCoordinateFrame}{#1}
    \tikzset{
      main_orientation/.style = {
        x = {
          (
            {transformrotmainX(\MCF@a,\MCF@b,\MCF@g,1,0,0) * 1cm},
            {transformrotmainY(\MCF@a,\MCF@b,\MCF@g,1,0,0) * 1cm}
          )
        },
        y = {
          (
            {transformrotmainX(\MCF@a,\MCF@b,\MCF@g,0,1,0) * 1cm},
            {transformrotmainY(\MCF@a,\MCF@b,\MCF@g,0,1,0) * 1cm}
          )
        },
        z = {
          (
            {transformrotmainZ(\MCF@a,\MCF@b,\MCF@g,1,0,0) * 1cm},
            {transformrotmainZ(\MCF@a,\MCF@b,\MCF@g,0,1,0) * 1cm}
          )
        }
      }
    }
}
