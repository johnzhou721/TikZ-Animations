% rotation matrix about arbitrary axis

% Rodrigues: X‐component
\pgfmathdeclarefunction{rodrX}{7}{%
  \pgfmathparse{%
    % unpack
    % #1=theta, #2=ux, #3=uy, #4=uz, #5=x, #6=y, #7=z
    cos(#1)*#5
      + (1-cos(#1))*(#2*(#2*#5 + #3*#6 + #4*#7))
      + sin(#1)*((#3*#7 - #4*#6))
  }%
}

% Rodrigues: Y‐component
\pgfmathdeclarefunction{rodrY}{7}{%
  \pgfmathparse{%
    cos(#1)*#6
      + (1-cos(#1))*(#3*(#2*#5 + #3*#6 + #4*#7))
      + sin(#1)*((#4*#5 - #2*#7))
  }%
}

% Rodrigues: Z‐component
\pgfmathdeclarefunction{rodrZ}{7}{%
  \pgfmathparse{%
    cos(#1)*#7
      + (1-cos(#1))*(#4*(#2*#5 + #3*#6 + #4*#7))
      + sin(#1)*((#2*#6 - #3*#5))
  }%
}
\tikzset{
  rodrigues_orientation/.code n args = {4}{
    \begingroup
      \pgfmathsetmacro{\xx}{rodrX(#1,#2,#3,#4,1,0,0)}
      \pgfmathsetmacro{\xy}{rodrY(#1,#2,#3,#4,1,0,0)}
      \pgfmathsetmacro{\yx}{rodrX(#1,#2,#3,#4,0,1,0)}
      \pgfmathsetmacro{\yy}{rodrY(#1,#2,#3,#4,0,1,0)}
      \pgfmathsetmacro{\zx}{rodrX(#1,#2,#3,#4,0,0,1)}
      \pgfmathsetmacro{\zy}{rodrY(#1,#2,#3,#4,0,0,1)}
      \edef\temp{
        \noexpand\pgfkeysalso{
          x = {
            (\xx cm,\xy cm)
          },
          y = {
            (\yx cm,\yy cm)
          },
          z = {
            (\zx cm,\zy cm)
          }
        }
      }
      \expandafter
    \endgroup\temp
  }
}
